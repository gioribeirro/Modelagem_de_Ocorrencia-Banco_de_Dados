-- Execute uma unica vez em um servidor sem db_52dp_v3 preexistente.
CREATE DATABASE db_52dp_v3 CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE db_52dp_v3;

-- 1. ESTRUTURA: 11 TABELAS
CREATE TABLE UNIDADE_POLICIAL (
 id_unidade INT AUTO_INCREMENT PRIMARY KEY,
 nome VARCHAR(120) NOT NULL,
 endereco VARCHAR(150) NOT NULL,
 tipo_unidade VARCHAR(60) NOT NULL,
 cidade VARCHAR(80) NOT NULL
);
CREATE TABLE PESSOA (
 id_pessoa INT AUTO_INCREMENT PRIMARY KEY,
 nome_ficticio VARCHAR(120) NOT NULL,
 documento_ficticio VARCHAR(30) NOT NULL,
 observacao VARCHAR(200) NULL,
 contato_ficticio VARCHAR(20) NOT NULL
);
CREATE TABLE SERVIDOR (
 id_servidor INT AUTO_INCREMENT PRIMARY KEY,
 id_pessoa INT NOT NULL UNIQUE,
 id_unidade INT NOT NULL,
 cargo VARCHAR(60) NULL, -- obrigatoriedade nao declarada no dicionario (‡)
 matricula_ficticia VARCHAR(20) NOT NULL,
 CONSTRAINT fk_servidor_pessoa FOREIGN KEY (id_pessoa) REFERENCES PESSOA(id_pessoa),
 CONSTRAINT fk_servidor_unidade FOREIGN KEY (id_unidade) REFERENCES UNIDADE_POLICIAL(id_unidade),
 CONSTRAINT chk_servidor_cargo CHECK (cargo IS NULL OR cargo IN ('Delegado','Escrivão','Investigador/Agente'))
);
CREATE TABLE ADVOGADO (
 id_oab VARCHAR(20) PRIMARY KEY,
 id_pessoa INT NOT NULL UNIQUE,
 CONSTRAINT fk_advogado_pessoa FOREIGN KEY (id_pessoa) REFERENCES PESSOA(id_pessoa)
);
CREATE TABLE NATUREZA_FATO (
 id_natureza INT AUTO_INCREMENT PRIMARY KEY,
 descricao VARCHAR(100) NOT NULL,
 gravidade VARCHAR(20) NOT NULL
);
CREATE TABLE OCORRENCIA (
 id_ocorrencia INT AUTO_INCREMENT PRIMARY KEY,
 id_unidade INT NOT NULL,
 id_natureza INT NOT NULL,
 id_servidor INT NULL, -- obrigatoriedade nao declarada (‡)
 canal_registro VARCHAR(20) NOT NULL,
 data_registro DATETIME NOT NULL,
 localidade VARCHAR(120) NOT NULL,
 status VARCHAR(30) NULL, -- obrigatoriedade nao declarada (‡)
 relato_fatos TEXT NOT NULL,
 CONSTRAINT fk_ocorrencia_unidade FOREIGN KEY (id_unidade) REFERENCES UNIDADE_POLICIAL(id_unidade),
 CONSTRAINT fk_ocorrencia_natureza FOREIGN KEY (id_natureza) REFERENCES NATUREZA_FATO(id_natureza),
 CONSTRAINT fk_ocorrencia_servidor FOREIGN KEY (id_servidor) REFERENCES SERVIDOR(id_servidor),
 CONSTRAINT chk_ocorrencia_status CHECK (status IS NULL OR status IN ('Em análise','Em investigação','Convertida em inquérito','Cancelada')),
 CONSTRAINT chk_ocorrencia_canal CHECK (canal_registro IN ('Presencial','Online'))
);
CREATE TABLE ENVOLVIMENTO (
 id_pessoa INT NOT NULL,
 id_ocorrencia INT NOT NULL,
 id_oab VARCHAR(20) NULL,
 papel VARCHAR(30) NOT NULL,
 PRIMARY KEY (id_pessoa,id_ocorrencia),
 CONSTRAINT fk_envolvimento_pessoa FOREIGN KEY (id_pessoa) REFERENCES PESSOA(id_pessoa),
 CONSTRAINT fk_envolvimento_ocorrencia FOREIGN KEY (id_ocorrencia) REFERENCES OCORRENCIA(id_ocorrencia),
 CONSTRAINT fk_envolvimento_advogado FOREIGN KEY (id_oab) REFERENCES ADVOGADO(id_oab),
 CONSTRAINT chk_envolvimento_papel CHECK (papel <> 'Advogado')
);
CREATE TABLE INQUERITO (
 id_inquerito INT AUTO_INCREMENT PRIMARY KEY,
 id_ocorrencia INT NOT NULL UNIQUE,
 data_abertura DATE NOT NULL,
 data_conclusao DATE NULL,
 status VARCHAR(30) NOT NULL,
 prioridade VARCHAR(20) NOT NULL,
 CONSTRAINT fk_inquerito_ocorrencia FOREIGN KEY (id_ocorrencia) REFERENCES OCORRENCIA(id_ocorrencia),
 CONSTRAINT chk_datas_inquerito CHECK (data_conclusao IS NULL OR data_conclusao >= data_abertura)
);
CREATE TABLE DEPOIMENTO (
 id_depoimento INT AUTO_INCREMENT PRIMARY KEY,
 id_inquerito INT NOT NULL,
 id_pessoa INT NOT NULL,
 data_depoimento DATE NOT NULL,
 resumo_ficticio VARCHAR(255) NOT NULL,
 CONSTRAINT fk_depoimento_inquerito FOREIGN KEY (id_inquerito) REFERENCES INQUERITO(id_inquerito),
 CONSTRAINT fk_depoimento_pessoa FOREIGN KEY (id_pessoa) REFERENCES PESSOA(id_pessoa)
);
CREATE TABLE EVIDENCIA (
 id_evidencia INT AUTO_INCREMENT PRIMARY KEY,
 id_inquerito INT NOT NULL,
 tipo_evidencia VARCHAR(60) NOT NULL,
 descricao VARCHAR(200) NOT NULL,
 data_apreensao DATE NOT NULL,
 status_custodia VARCHAR(30) NOT NULL,
 CONSTRAINT fk_evidencia_inquerito FOREIGN KEY (id_inquerito) REFERENCES INQUERITO(id_inquerito)
);
CREATE TABLE MOVIMENTACAO (
 id_movimentacao INT AUTO_INCREMENT PRIMARY KEY,
 id_inquerito INT NOT NULL,
 id_servidor INT NULL, -- obrigatoriedade nao declarada (‡)
 data_movimentacao DATE NOT NULL,
 tipo_movimentacao VARCHAR(60) NOT NULL,
 descricao VARCHAR(200) NOT NULL,
 CONSTRAINT fk_movimentacao_inquerito FOREIGN KEY (id_inquerito) REFERENCES INQUERITO(id_inquerito),
 CONSTRAINT fk_movimentacao_servidor FOREIGN KEY (id_servidor) REFERENCES SERVIDOR(id_servidor)
);

-- 3. CONSULTAS E JOINS
SELECT * FROM OCORRENCIA;
SELECT o.id_ocorrencia,o.canal_registro,o.status,n.descricao AS natureza,
 u.nome AS unidade,p.nome_ficticio AS servidor_registro
FROM OCORRENCIA o
JOIN NATUREZA_FATO n ON n.id_natureza=o.id_natureza
JOIN UNIDADE_POLICIAL u ON u.id_unidade=o.id_unidade
LEFT JOIN SERVIDOR s ON s.id_servidor=o.id_servidor
LEFT JOIN PESSOA p ON p.id_pessoa=s.id_pessoa
ORDER BY o.id_ocorrencia;
-- Pessoas envolvidas e advogado atualmente vinculado, quando houver
SELECT o.id_ocorrencia,p.nome_ficticio AS pessoa,e.papel,
 a.id_oab,pa.nome_ficticio AS advogado
FROM ENVOLVIMENTO e
JOIN OCORRENCIA o ON o.id_ocorrencia=e.id_ocorrencia
JOIN PESSOA p ON p.id_pessoa=e.id_pessoa
LEFT JOIN ADVOGADO a ON a.id_oab=e.id_oab
LEFT JOIN PESSOA pa ON pa.id_pessoa=a.id_pessoa
ORDER BY o.id_ocorrencia,p.nome_ficticio;
-- Depoimentos por inquérito
SELECT d.id_depoimento,i.id_inquerito,p.nome_ficticio,d.data_depoimento,d.resumo_ficticio
FROM DEPOIMENTO d
JOIN INQUERITO i ON i.id_inquerito=d.id_inquerito
JOIN PESSOA p ON p.id_pessoa=d.id_pessoa;
-- Auditoria demonstrativa de movimentações (nao e log de acesso)
SELECT m.id_movimentacao,m.data_movimentacao,m.tipo_movimentacao,
 i.id_inquerito,p.nome_ficticio AS servidor,m.descricao
FROM MOVIMENTACAO m
JOIN INQUERITO i ON i.id_inquerito=m.id_inquerito
LEFT JOIN SERVIDOR s ON s.id_servidor=m.id_servidor
LEFT JOIN PESSOA p ON p.id_pessoa=s.id_pessoa;

SELECT n.descricao AS natureza,COUNT(*) AS total_ocorrencias
FROM OCORRENCIA o JOIN NATUREZA_FATO n ON n.id_natureza=o.id_natureza
GROUP BY n.id_natureza,n.descricao ORDER BY total_ocorrencias DESC;
SELECT canal_registro,COUNT(*) AS total_ocorrencias
FROM OCORRENCIA GROUP BY canal_registro ORDER BY total_ocorrencias DESC;
